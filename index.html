/** * BACKEND SISTEM PERSURATAN & ARSIP CLOUD SDN 3 NOGORAJI * Arsitektur: Brankas Multi-Akun Terisolasi (Isolated User Vaults) * Data disimpan langsung di Google Drive sehingga dapat diakses dari laptop & HP mana pun. */ function doGet(e) { try { var action = (e && e.parameter) ? e.parameter.action : ""; var targetFolder = getOrCreateFolder("Arsip Surat SDN 3 Nogoraji"); var dbFileName = "database_sdn3_cloud.json"; // 1. TARIK DATA CLOUD LENGKAP SAAT APLIKASI DIBUKA DI HP / LAPTOP if (action === "load_data" || action === "load_vault") { var files = targetFolder.getFilesByName(dbFileName); if (files.hasNext()) { var dbFile = files.next(); var content = dbFile.getBlob().getDataAsString(); return ContentService.createTextOutput(content) .setMimeType(ContentService.MimeType.JSON); } else { // Inisialisasi awal database cloud baru var initData = { status: "success", users: { "arifprasetyo1543@gmail.com": { email: "arifprasetyo1543@gmail.com", name: "Arif Prasetyo", role: "Operator Sekolah", password: "123", authenticatorEnabled: false }, "sdn3nogoraji@gmail.com": { email: "sdn3nogoraji@gmail.com", name: "Kasmin, S.Pd., M.M.", role: "Kepala Sekolah", password: "123", authenticatorEnabled: false } }, vaults: {} }; return ContentService.createTextOutput(JSON.stringify(initData)) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput("SERVER CLOUD GOOGLE DRIVE SDN 3 NOGORAJI ONLINE & SIAP DIGUNAKAN!") .setMimeType(ContentService.MimeType.TEXT); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var raw = e.postData ? e.postData.contents : ""; if (!raw) throw new Error("Tidak ada data yang diterima."); var data = JSON.parse(raw); var targetFolder = getOrCreateFolder("Arsip Surat SDN 3 Nogoraji"); var dbFileName = "database_sdn3_cloud.json"; // 2. UNGGAH BERKAS PDF FISIK KE GOOGLE DRIVE if (data.action === "upload_pdf") { var decoded = Utilities.base64Decode(data.base64); var blob = Utilities.newBlob(decoded, "application/pdf", data.fileName || "Berkas_Arsip.pdf"); var file = targetFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return ContentService.createTextOutput(JSON.stringify({ status: "success", fileUrl: file.getUrl(), fileId: file.getId(), fileName: file.getName() })).setMimeType(ContentService.MimeType.JSON); } // 3. SINKRONISASI BRANKAS DATA PER AKUN (ISOLATED VAULT) if (data.action === "sync_user_vault") { var emailKey = String(data.email || "").toLowerCase().trim(); if (!emailKey) throw new Error("Email akun wajib dicantumkan."); var files = targetFolder.getFilesByName(dbFileName); var db = { users: {}, vaults: {} }; if (files.hasNext()) { var existingFile = files.next(); try { db = JSON.parse(existingFile.getBlob().getDataAsString()); } catch(e) { db = { users: {}, vaults: {} }; } } if (!db.users) db.users = {}; if (!db.vaults) db.vaults = {}; // Perbarui data pengguna if (data.account) { db.users[emailKey] = data.account; } // Perbarui brankas spesifik milik akun ini saja if (data.vault) { db.vaults[emailKey] = { kop: data.vault.kop || null, pegawai: data.vault.pegawai || [], arsip: data.vault.arsip || [], lastUpdated: new Date().toISOString() }; } db.status = "success"; var newContent = JSON.stringify(db, null, 2); if (files.hasNext()) { existingFile.setContent(newContent); } else { var newFile = targetFolder.createFile(dbFileName, newContent, "application/json"); newFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data brankas akun " + emailKey + " berhasil tersimpan di Google Drive!" })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Aksi tidak dikenali." })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function getOrCreateFolder(folderName) { var folders = DriveApp.getFoldersByName(folderName); if (folders.hasNext()) { return folders.next(); } else { return DriveApp.createFolder(folderName); } }
